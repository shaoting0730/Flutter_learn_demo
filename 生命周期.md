## didChangeDependencies 调用时机<br/>
> `didChangeDependencies` 在 两个时机 被调用：<br/>
> 1、首次创建时，在 initState() 之后、build() 之前 <br/>
> 2、依赖的 InheritedWidget 发生变化时（如 Theme、Provider、MediaQuery 等）<br/>
父组件重建、父级结构变化、Widget 类型变化等都 不会 触发此方法。
## didUpdateWidget 调用时机<br/>
> `didUpdateWidget` 在 父组件重建并传入新的 Widget 实例，且 State 对象被复用 时调用。<br/>
> 必须同时满足三个条件：<br/>
> 父组件执行了 build() 生成了新 Widget<br/>
> 新旧 Widget 的 runtimeType 相同<br/>
> 新旧 Widget 的 Key 相同（或都没有 Key）<br/>
如果子 Widget 使用了 const，则不会触发此方法。<br/>

```
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Flutter Demo',
      theme: ThemeData(colorScheme: .fromSeed(seedColor: Colors.deepPurple)),
      home: const MyHomePage(title: 'Flutter Demo Home Page'),
    );
  }
}

class MyHomePage extends StatefulWidget {
  const MyHomePage({super.key, required this.title});
  final String title;

  @override
  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {
  int _counter = 0;
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  initState() {
    super.initState();
    print('-----------------------------------------根root中的initState-----------------------------------------');
  }

  @override
  dispose() {
    super.dispose();
    print('根root中的dispose');
  }

  @override
  didUpdateWidget(MyHomePage oldWidget) {
    super.didUpdateWidget(oldWidget);
    print('根root中的didUpdateWidget');
  }

  @override
  didChangeDependencies() {
    super.didChangeDependencies();
    print('根root中的didChangeDependencies');
  }

  @override
  Widget build(BuildContext context) {
    print('根root中的build');
    return Scaffold(
      appBar: AppBar(backgroundColor: Theme.of(context).colorScheme.inversePrimary, title: Text(widget.title)),
      body: Center(
        child: Column(
          mainAxisAlignment: .center,
          children: [
            const Text('根界面'),
            Text('根界面counter: $_counter', style: Theme.of(context).textTheme.headlineMedium),
            Rootchild(),
          ],
        ),
      ),
      floatingActionButton: FloatingActionButton(onPressed: _incrementCounter, tooltip: 'Increment', child: const Icon(Icons.add)),
    );
  }
}

class Rootchild extends StatefulWidget {
  const Rootchild({super.key});

  @override
  State<Rootchild> createState() => _RootchildState();
}

class _RootchildState extends State<Rootchild> {
  int _counter = 0;
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  initState() {
    super.initState();
    print('-----------------------------------------子界面中的initState-----------------------------------------');
  }

  @override
  dispose() {
    super.dispose();
    print('子界面中的dispose');
  }

  @override
  didUpdateWidget(Rootchild oldWidget) {
    super.didUpdateWidget(oldWidget);
    print('子界面中的didUpdateWidget');
  }

  @override
  didChangeDependencies() {
    super.didChangeDependencies();
    print('子界面中的didChangeDependencies');
  }

  @override
  Widget build(BuildContext context) {
    print('子界面中的build');
    return Container(
      padding: EdgeInsets.all(10),
      color: Colors.red,
      child: Column(
        children: [
          Text('子界面'),
          Text('子界面counter: $_counter', style: Theme.of(context).textTheme.headlineMedium),
          InkWell(
            onTap: _incrementCounter,
            child: Text('子界面+', style: Theme.of(context).textTheme.headlineMedium),
          ),
          InkWell(
            onTap: () {
              Navigator.push(context, MaterialPageRoute(builder: (context) => OnePage()));
            },
            child: Text('push到OnePage', style: Theme.of(context).textTheme.headlineMedium),
          ),
        ],
      ),
    );
  }
}

class OnePage extends StatefulWidget {
  const OnePage({super.key});

  @override
  State<OnePage> createState() => _OnePageState();
}

class _OnePageState extends State<OnePage> {
  int _counter = 0;
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  initState() {
    super.initState();
    print('-----------------------------------------OnePage界面中的initState-----------------------------------------');
  }

  @override
  dispose() {
    super.dispose();
    print('OnePage界面中的dispose');
  }

  @override
  didUpdateWidget(OnePage oldWidget) {
    super.didUpdateWidget(oldWidget);
    print('OnePage界面中的didUpdateWidget');
  }

  @override
  didChangeDependencies() {
    super.didChangeDependencies();
    print('OnePage界面中的didChangeDependencies');
  }

  @override
  Widget build(BuildContext context) {
    print('OnePage界面中的build');
    return Scaffold(
      body: Container(
        padding: EdgeInsets.all(10),
        color: Colors.orange,
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          crossAxisAlignment: CrossAxisAlignment.center,
          children: [
            Text('OnePage界面'),
            Text('OnePage界面counter: $_counter', style: Theme.of(context).textTheme.headlineMedium),
            InkWell(
              onTap: _incrementCounter,
              child: Text('OnePage界面+', style: Theme.of(context).textTheme.headlineMedium),
            ),
            InkWell(
              onTap: () {
                Navigator.push(context, MaterialPageRoute(builder: (context) => TwoPage()));
              },
              child: Text('push到TwoPage', style: Theme.of(context).textTheme.headlineMedium),
            ),
            InkWell(
              onTap: () {
                Navigator.pop(context);
              },
              child: Text('pop到Main', style: Theme.of(context).textTheme.headlineMedium),
            ),
          ],
        ),
      ),
    );
  }
}

class TwoPage extends StatefulWidget {
  const TwoPage({super.key});

  @override
  State<TwoPage> createState() => _TwoPageState();
}

class _TwoPageState extends State<TwoPage> {
  int _counter = 0;
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  @override
  initState() {
    super.initState();
    print('-----------------------------------------TwoPage界面中的initState-----------------------------------------');
  }

  @override
  dispose() {
    super.dispose();
    print('TwoPage界面中的dispose');
  }

  @override
  didUpdateWidget(TwoPage oldWidget) {
    super.didUpdateWidget(oldWidget);
    print('TwoPage界面中的didUpdateWidget');
  }

  @override

  }

  Widget build(BuildContext context) {

    return Scaffold(
      body: Container(
        padding: EdgeInsets.all(10),

          crossAxisAlignment: CrossAxisAlignment.center,
          children: [

            Text('TwoPage界面中counter: $_counter', style: Theme.of(context).textTheme.headlineMedium),

              onTap: () {

              },

          ],
        ),
      ),
    );
  }
}


```
四个`StatefulWidget `: `MyHomePage ` `Rootchild ` `OnePage ` `TwoPage ` <br/>
`MyHomePage ` 是根Widget, `Rootchild ` 是  `MyHomePage ` 的子组件, `OnePage ` 和 `TwoPage ` 是新的组件  <br/>
四个`StatefulWidget `都写有四个生命周期方法:
测试步骤: <br/>
进入界面、点击主界面的[+]号三次,然后点击子组件的[子界面+]号三次,再点击按钮[push到OnePage]到了 `OnePage` ,点击
`OnePage界面+`三次,然后再点击按钮[push到TwoPage]到了 `TwoPage`,到了 `TwoPage`,点击按钮 `TwoPage界面中+1`三次,
然后依次点击按钮 `pop到OnePage` , `pop到Main`.
